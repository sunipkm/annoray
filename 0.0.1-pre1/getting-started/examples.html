Examples#

Each example exists in Rust and in Python, and both print the same lines. The sources are in annoray/examples and annoray-python/examples:

cargo run --release --example quickstart   # in annoray/
python examples/quickstart.py              # in annoray-python/

Quickstart#

Building arrays and datasets, selecting, arithmetic with units, reductions.

"""annoray quick start: build, inspect, select, compute with units, reduce.

After xarray's "Quick overview". The Rust twin is `annoray/examples/quickstart.rs`; both
print the same lines.

    python examples/quickstart.py
"""

import numpy as np

import annoray as ar


def fmt(v):
    return f"{float(v):.3f}"


def show(label, values):
    print(f"{label}: [" + ", ".join(fmt(v) for v in np.ravel(values)) + "]")


# A DataArray: values, dimension names, coordinates and a unit.
da = ar.DataArray(
    np.arange(6.0).reshape(2, 3),
    dims=("x", "y"),
    coords={"x": [10.0, 20.0], "y": [1.0, 2.0, 3.0]},
    name="height",
    units="m",
)
print(f"dims: {', '.join(da.dims)}")
print("sizes: " + ", ".join(f"{k}={n}" for k, n in da.sizes.items()))
print(f"units: {da.units}")

# Selection by position (isel) and by label (sel); label slices include both ends.
show("isel x=0", da.isel(x=0).values)
show("sel x=20", da.sel(x=20.0).values)
show("sel y=1.5:3", da.sel(y=slice(1.5, 3.0)).values)
show("sel y~2.4 (nearest)", da.sel(y=2.4, method="nearest").values)

# Arithmetic carries units, and refuses what does not make sense.
seconds = ar.DataArray(2.0, units="s")
print(f"da * 2 -> {(da * 2.0).units}")
print(f"da * da -> {(da * da).units}")
print(f"da / 2 s -> {(da / seconds).units}")
try:
    _ = da + seconds
except ar.UnitError as e:
    print(f"da + 2 s -> error: {e}")

# Reductions, by dimension name.
show("mean over x", da.mean("x").values)
show("std over y (ddof=1)", da.std("y", ddof=1).values)
show("sum", da.sum().values)

# Unit conversion.
show("in cm", da.to("cm").values)

# A Dataset: variables sharing dimensions; arithmetic broadcasts by name.
ds = ar.Dataset(
    {"height": da, "width": ("y", np.array([1.0, 2.0, 4.0]), {"units": "m"})},
    attrs={"title": "quick start"},
)
area = ds["height"] * ds["width"]
print(f"area units: {area.units}, dims: {', '.join(area.dims)}")
show("area", area.values)
print(ds)

//! annoray quick start: build, inspect, select, compute with units, reduce.
//!
//! After xarray's "Quick overview". The Python twin is
//! `annoray-python/examples/quickstart.py`; both print the same lines.
//!
//! ```text
//! cargo run --example quickstart
//! ```
// annoray's operators return `Result` (unit and shape errors are values, never panics); the
// workspace's `arithmetic_side_effects` lint cannot tell.
#![allow(clippy::arithmetic_side_effects)]

use std::error::Error;

use annoray::ndarray::{ArrayD, IxDyn, array};
use annoray::{
    DataArray, DataArrayBase, Dataset, ReduceOptions, Reduction, Repr, Variable, isel, sel,
};

/// Values with three decimals, as the Python twin prints them.
fn show<S: Repr>(label: &str, da: &DataArrayBase<S>) -> Result<(), annoray::CoreError> {
    let v: Vec<String> = da.to_f64()?.iter().map(|v| format!("{v:.3}")).collect();
    println!("{label}: [{}]", v.join(", ").replace("NaN", "nan"));
    Ok(())
}

fn main() -> Result<(), Box<dyn Error>> {
    // A DataArray: values, dimension names, coordinates and a unit.
    let data = ArrayD::from_shape_vec(IxDyn(&[2, 3]), (0..6).map(f64::from).collect())?;
    let da = DataArray::from_array(["x", "y"], data)?
        .with_dim_coord("x", array![10.0, 20.0])?
        .with_dim_coord("y", array![1.0, 2.0, 3.0])?
        .with_name("height")
        .with_unit("m".parse()?);
    println!("dims: {}", da.dims().join(", "));
    let sizes: Vec<String> = da.sizes().map(|(k, n)| format!("{k}={n}")).collect();
    println!("sizes: {}", sizes.join(", "));
    println!("units: {}", da.unit().ok_or("no unit")?);

    // Selection by position (isel) and by label (sel); label slices include both ends.
    show("isel x=0", &isel!(da[x = 0])?)?;
    show("sel x=20", &sel!(da[x = 20.0])?)?;
    show("sel y=1.5:3", &sel!(da[y = 1.5:3.0])?)?;
    show("sel y~2.4 (nearest)", &sel!(da[y = nearest(2.4)])?)?;

    // Arithmetic carries units, and refuses what does not make sense.
    let seconds = DataArray::scalar(2.0)?.with_unit("s".parse()?);
    let unit = |r: &DataArray| r.unit().map(ToString::to_string).unwrap_or_default();
    println!("da * 2 -> {}", unit(&(&da * 2.0)?));
    println!("da * da -> {}", unit(&(&da * &da)?));
    println!("da / 2 s -> {}", unit(&(&da / &seconds)?));
    match &da + &seconds {
        Ok(_) => println!("da + 2 s -> (no error?)"),
        Err(e) => println!("da + 2 s -> error: {e}"),
    }

    // Reductions, by dimension name.
    show("mean over x", &da.mean("x")?)?;
    let opts = ReduceOptions::default().ddof(1);
    show(
        "std over y (ddof=1)",
        &da.reduce(Reduction::Std, "y", &opts)?,
    )?;
    show("sum", &da.sum(..)?)?;

    // Unit conversion.
    show("in cm", &da.to_unit(&"cm".parse()?)?)?;

    // A Dataset: variables sharing dimensions; arithmetic broadcasts by name.
    let mut ds = Dataset::new().with_attrs([("title", "quick start")]);
    ds.add_array("height", da)?;
    ds.add_var(
        "width",
        Variable::new(["y"], array![1.0, 2.0, 4.0].into_dyn())?.with_unit("m".parse()?),
    )?;
    let area = (&ds.get("height")? * &ds.get("width")?)?;
    println!(
        "area units: {}, dims: {}",
        unit(&area),
        area.dims().join(", ")
    );
    show("area", &area)?;
    println!("{ds}");
    Ok(())
}

Quantities#

Scalars with units, dimension-checked at run time in Python and at compile time in Rust.

"""Quantities: numbers with units, their dimensions explicit and checked.

The Rust twin (`annoray/examples/quantities.rs`) checks dimensions when it compiles; this
one when it runs. Both print the same lines.

    python examples/quantities.py
"""

import annoray as ar
from annoray import Quantity
from annoray.dims import Area, Dim, Force, Length, Mass, Temperature, Time, Velocity, checked


class Jerk(Dim, dimension=Length / Time**3):
    """Rate of change of acceleration, composed from SI dimensions.

    (Rust: `dimension! { pub Jerk = Length / Time^3; }`.)"""


def yes(b):
    return "yes" if b else "no"


@checked
def speed(d: Quantity[Length], t: Quantity[Time]) -> Quantity[Velocity]:
    return d / t


# A length over a time is a velocity: checked going in and coming out.
d = Length(2.5, "km")
t = Time(30.0, "min")
v = speed(d, t)
print(f"speed: {v:.3f}")
print(f"speed: {v.to('m/s'):.3f}")

# A velocity over a time twice is a jerk. (Dividing once more would be m s-4: the Rust twin
# does not compile; here `Jerk.check` raises.)
s = t.to("s")
j = Jerk.check(v.to("m/s") / s / s)
print(f"jerk unit: {j.units}")
print(f"jerk: {j.to('um s-3'):.3f}")

# Physics reads like physics.
m = Mass(1200.0, "kg")
car = Velocity(27.8, "m/s")
e = m * car * car * 0.5
print(f"kinetic energy: {e.to('kJ'):.3f}")
p = Force(500.0, "N") / Area(25.0, "cm2")
print(f"pressure: {p.to('kPa'):.3f}")

# Units are values: same dimension, different scales need an explicit conversion.
km = Length(1.0, "km")
one_m = Length(1.0, "m")
try:
    _ = km + one_m
    print("1 km + 1 m -> (no error?)")
except ar.UnitError as err:
    print(f"1 km + 1 m -> error: {err}")
print(f"1 km + 1 m -> {km + one_m.to('km'):.3f}")
print(f"1 km == 1000 m: {yes(km == Length(1000.0, 'm'))}")

# Absolute temperatures: differences and conversions work; products do not.
warm = Temperature(20.0, "degC")
cool = Temperature(10.0, "degC")
print(f"20 degC - 10 degC = {warm - cool:.1f}")
print(f"20 degC = {warm.to('degF'):.1f}")
try:
    _ = warm * 2.0
    failed = False
except ar.UnitError:
    failed = True
print(f"2 x 20 degC -> error: {yes(failed)}")

# A unit known only at run time (from a file, say).
read = ar.Quantity(12.0, "m s-1")
print(f"is a velocity: {yes(isinstance(read, Velocity))}")
print(f"is a length: {yes(isinstance(read, Length))}")

//! Quantities: numbers with units, their dimensions checked by the compiler.
//!
//! The Python twin (`annoray-python/examples/quantities.py`) does the same with quantities
//! checked at run time; both print the same lines.
//!
//! ```text
//! cargo run --example quantities
//! ```
// Quantity operators return `Result` (unit errors are values, never panics), which the
// workspace's `arithmetic_side_effects` lint cannot tell.
#![allow(clippy::arithmetic_side_effects)]

use std::error::Error;

use annoray::dims::{Area, Energy, Force, Length, Mass, Pressure, Temperature, Time, Velocity};
use annoray::{Dyn, Quantity, dimension};

dimension! {
    /// Rate of change of acceleration, composed from SI dimensions.
    pub Jerk = Length / Time^3;
}

fn yes(b: bool) -> &'static str {
    if b { "yes" } else { "no" }
}

fn main() -> Result<(), Box<dyn Error>> {
    // A length over a time is a velocity: the result's type says so.
    let d = Quantity::<Length>::new(2.5, "km")?;
    let t = Quantity::<Time>::new(30.0, "min")?;
    let v: Quantity<Velocity> = (&d / &t)?;
    println!("speed: {v:.3}");
    println!("speed: {:.3}", v.to("m/s")?);

    // Dimensions of your own, composed from the SI ones. A velocity over a time twice is a
    // jerk; dividing once more would not compile (it would be m s-4, not a jerk).
    let s = t.to("s")?;
    let j: Quantity<Jerk> = (&(&v.to("m/s")? / &s)? / &s)?;
    println!("jerk unit: {}", j.unit());
    println!("jerk: {:.3}", j.to("um s-3")?);

    // Physics reads like physics.
    let m = Quantity::<Mass>::new(1200.0, "kg")?;
    let car = Quantity::<Velocity>::new(27.8, "m/s")?;
    let e: Quantity<Energy> = ((&(&m * &car)? * &car)? * 0.5)?;
    println!("kinetic energy: {:.3}", e.to("kJ")?);
    let f = Quantity::<Force>::new(500.0, "N")?;
    let a = Quantity::<Area>::new(25.0, "cm2")?;
    let p: Quantity<Pressure> = (&f / &a)?;
    println!("pressure: {:.3}", p.to("kPa")?);

    // Units are values: same dimension, different scales need an explicit conversion.
    let km = Quantity::<Length>::new(1.0, "km")?;
    let one_m = Quantity::<Length>::new(1.0, "m")?;
    match &km + &one_m {
        Ok(_) => println!("1 km + 1 m -> (no error?)"),
        Err(e) => println!("1 km + 1 m -> error: {e}"),
    }
    println!("1 km + 1 m -> {:.3}", (&km + &one_m.to("km")?)?);
    println!(
        "1 km == 1000 m: {}",
        yes(km == Quantity::<Length>::new(1000.0, "m")?)
    );
    // (`km + t` does not compile: a length and a time do not add.)

    // Absolute temperatures: differences and conversions work; products do not.
    let warm = Quantity::<Temperature>::new(20.0, "degC")?;
    let cool = Quantity::<Temperature>::new(10.0, "degC")?;
    println!("20 degC - 10 degC = {:.1}", (&warm - &cool)?);
    println!("20 degC = {:.1}", warm.to("degF")?);
    println!("2 x 20 degC -> error: {}", yes((&warm * 2.0).is_err()));

    // A unit known only at run time (from a file, say): checked when cast.
    let read: Quantity<Dyn> = Quantity::new(12.0, "m s-1")?;
    println!(
        "is a velocity: {}",
        yes(read.clone().cast::<Velocity>().is_ok())
    );
    println!("is a length: {}", yes(read.cast::<Length>().is_ok()));
    Ok(())
}

Weather#

Grouping, resampling and climatologies on a synthetic weather record.

"""Toy weather data: climatologies, anomalies, resampling, rolling means, missing values.

After xarray's "Toy weather data" example, with deterministic noise instead of random
numbers so the Rust twin (`annoray/examples/weather.rs`) prints the same lines.

    python examples/weather.py
"""

import numpy as np

import annoray as ar


def fmt(v):
    return f"{float(v):.3f}"


def show(label, values):
    print(f"{label}: [" + ", ".join(fmt(v) for v in np.ravel(values)) + "]")


# Two years of daily minimum and maximum temperatures at three places.
times = ar.date_range("2000-01-01", 731, "D")
doy = ar.DataArray(times, dims="time").dt.dayofyear.values
annual = np.sin(2 * np.pi * (doy / 365.25 - 0.28))
base = 10 + 15 * annual[:, None]
i = np.arange(len(times))[:, None]
j = np.arange(3)[None, :]
tmin = base + 3 * np.sin(0.7 * (7 * i + 13 * j))
tmax = base + 10 + 3 * np.sin(0.7 * (7 * i + 13 * (j + 5)))
tmin[::50, 0] = np.nan  # some missing readings at the first place

ds = ar.Dataset(
    {
        "tmin": (("time", "location"), tmin, {"units": "degC"}),
        "tmax": (("time", "location"), tmax, {"units": "degC"}),
    },
    coords={"time": times, "location": np.array(["IA", "IN", "IL"], dtype=object)},
)
print("sizes: " + ", ".join(f"{k}={n}" for k, n in ds.sizes.items()))

# Missing values: counted, skipped by reductions, or filled.
show("missing tmin per place", ds["tmin"].isnull().sum("time").values)
show("mean tmin (skipping NaN)", ds["tmin"].mean("time").values)

# Monthly climatology, and each day's anomaly from it.
clim = ds["tmin"].groupby("time.month").mean()
show("IA tmin by month", clim.sel(location="IA").values)
anom = ds["tmin"].groupby("time.month") - clim
show(
    "IN anomaly, first week of June 2000",
    anom.sel(location="IN", time=slice("2000-06-01", "2000-06-07")).values,
)

# Resampling to monthly means, and a centred 7-day rolling mean.
monthly = ds["tmax"].resample(time="MS").mean()
show("IN monthly tmax, 2000 H1", monthly.sel(location="IN").isel(time=slice(0, 6)).values)
weekly = ds["tmax"].rolling(time=7, center=True).mean()
show("IL 7-day tmax, first days", weekly.sel(location="IL").isel(time=slice(0, 6)).values)

# The daily range, and units.
show("mean daily range", (ds["tmax"] - ds["tmin"]).mean("time").values)
show("mean tmax in degF", ds["tmax"].mean("time").to("degF").values)
show(
    "filled tmin, IA, days 0-2",
    ds["tmin"].fillna(ar.Quantity(-99.0, "degC")).sel(location="IA").isel(time=slice(0, 3)).values,
)

//! Toy weather data: climatologies, anomalies, resampling, rolling means, missing values.
//!
//! After xarray's "Toy weather data" example, with deterministic noise instead of random
//! numbers so the Python twin (`annoray-python/examples/weather.py`) prints the same lines.
//!
//! ```text
//! cargo run --example weather
//! ```
// annoray's operators return `Result` (never panic), which `arithmetic_side_effects` cannot
// tell; the casts are small loop counters.
#![allow(clippy::cast_precision_loss, clippy::arithmetic_side_effects)]

use std::error::Error;
use std::f64::consts::PI;

use annoray::ndarray::{Array1, Array2};
use annoray::{
    DataArray, DataArrayBase, Dataset, Repr, RollingOptions, Variable, date_range, isel, sel,
};

/// Values with three decimals, as the Python twin prints them.
fn show<S: Repr>(label: &str, da: &DataArrayBase<S>) -> Result<(), annoray::CoreError> {
    let v: Vec<String> = da.to_f64()?.iter().map(|v| format!("{v:.3}")).collect();
    println!("{label}: [{}]", v.join(", ").replace("NaN", "nan"));
    Ok(())
}

fn main() -> Result<(), Box<dyn Error>> {
    // Two years of daily minimum and maximum temperatures at three places.
    let time = Variable::new(["time"], date_range("2000-01-01", 731, "D")?.into_dyn())?;
    let doy = time.dt("dayofyear")?.to_f64()?;
    let base: Vec<f64> = doy
        .iter()
        .map(|d| 10.0 + 15.0 * (2.0 * PI * (d / 365.25 - 0.28)).sin())
        .collect();
    let at = |i: usize| base.get(i).copied().unwrap_or(f64::NAN);
    let noise = |i: usize, j: usize| 3.0 * (0.7 * (7 * i + 13 * j) as f64).sin();
    let tmin = Array2::from_shape_fn((base.len(), 3), |(i, j)| {
        // Some missing readings at the first place.
        if j == 0 && i % 50 == 0 {
            f64::NAN
        } else {
            at(i) + noise(i, j)
        }
    });
    let tmax = Array2::from_shape_fn((base.len(), 3), |(i, j)| at(i) + 10.0 + noise(i, j + 5));

    let mut ds = Dataset::new();
    ds.add_coord("time", time)?;
    let places: Array1<String> = ["IA", "IN", "IL"].map(String::from).to_vec().into();
    ds.add_coord("location", Variable::new(["location"], places.into_dyn())?)?;
    for (name, values) in [("tmin", tmin), ("tmax", tmax)] {
        let v = Variable::new(["time", "location"], values.into_dyn())?.with_unit("degC".parse()?);
        ds.add_var(name, v)?;
    }
    let sizes: Vec<String> = ds.sizes().iter().map(|(k, n)| format!("{k}={n}")).collect();
    println!("sizes: {}", sizes.join(", "));
    let tmin: DataArray = ds.get("tmin")?.to_owned();
    let tmax: DataArray = ds.get("tmax")?.to_owned();

    // Missing values: counted, skipped by reductions, or filled.
    show("missing tmin per place", &tmin.isnull()?.sum("time")?)?;
    show("mean tmin (skipping NaN)", &tmin.mean("time")?)?;

    // Monthly climatology, and each day's anomaly from it.
    let clim = tmin.groupby("time.month")?.mean()?;
    show("IA tmin by month", &sel!(clim[location = "IA"])?)?;
    let anom = (&tmin.groupby("time.month")? - &clim)?;
    show(
        "IN anomaly, first week of June 2000",
        &sel!(anom[location = "IN", time = "2000-06-01":"2000-06-07"])?,
    )?;

    // Resampling to monthly means, and a centred 7-day rolling mean.
    let monthly = tmax.resample("time", "MS")?.mean()?;
    let monthly = sel!(monthly[location = "IN"])?;
    show("IN monthly tmax, 2000 H1", &isel!(monthly[time = 0:6])?)?;
    let centred = RollingOptions::default().center(true);
    let weekly = tmax.rolling("time", 7, &centred)?.mean()?;
    let weekly = sel!(weekly[location = "IL"])?;
    show("IL 7-day tmax, first days", &isel!(weekly[time = 0:6])?)?;

    // The daily range, and units.
    show("mean daily range", &(&tmax - &tmin)?.mean("time")?)?;
    show(
        "mean tmax in degF",
        &tmax.mean("time")?.to_unit(&"degF".parse()?)?,
    )?;
    let filled = tmin.fillna(-99.0)?;
    let filled = sel!(filled[location = "IA"])?;
    show("filled tmin, IA, days 0-2", &isel!(filled[time = 0:3])?)?;
    Ok(())
}

Disk to disk#

Compressed netCDF out, lazy reads, and computations streamed from disk without loading the data.

"""Disk to disk: compressed netCDF out, lazy reads, streamed computation and reductions.

Data opened from files stays on disk: selections, arithmetic, unit conversion and
reductions read it chunk by chunk, so it never has to fit in memory. The Rust twin is
`annoray/examples/disk_to_disk.rs`; both print the same lines.

    python examples/disk_to_disk.py
"""

import shutil
import tempfile
from pathlib import Path

import numpy as np

import annoray as ar


def fmt(v):
    return f"{float(v):.3f}"


def yes(b):
    return "yes" if b else "no"


def show(label, values):
    print(f"{label}: [" + ", ".join(fmt(v) for v in np.ravel(values)) + "]")


tmp = Path(tempfile.mkdtemp(prefix="annoray-example-"))
try:
    # Two years of daily air temperature on a small grid, in kelvin.
    times = ar.date_range("2000-01-01", 731, "D")
    t = np.arange(731)[:, None, None]
    lat = np.linspace(-45.0, 45.0, 20)
    lon = np.arange(0.0, 360.0, 12.0)
    air = (
        288.0
        - 30.0 * np.abs(np.sin(np.radians(lat)))[None, :, None]
        + 8.0 * np.sin(2 * np.pi * t / 365.25)
        + np.cos(np.radians(lon))[None, None, :]
    )
    ds = ar.Dataset(
        {"air": (("time", "lat", "lon"), air, {"units": "K"})},
        coords={"time": times, "lat": lat, "lon": lon},
    )

    # Compressed, chunked netCDF-4; and one file per year.
    enc = {"air": {"zlib": True, "complevel": 4, "chunksizes": (30, 20, 30)}}
    ds.to_netcdf(tmp / "air.nc", encoding=enc)
    ds.isel(time=slice(0, 366)).to_netcdf(tmp / "air_2000.nc", encoding=enc)
    ds.isel(time=slice(366, 731)).to_netcdf(tmp / "air_2001.nc", encoding=enc)
    print("wrote air.nc, air_2000.nc, air_2001.nc")

    # Opening reads metadata only; values stay on disk until used.
    on_disk = ar.open_dataset(tmp / "air.nc")
    print(f"opened: lazy={yes(on_disk['air'].is_lazy)}")

    # Arithmetic and unit conversion stay lazy; writing computes chunk by chunk.
    celsius = on_disk["air"].to("degC")
    anomaly = celsius - celsius.mean("time")
    print(f"anomaly: lazy={yes(anomaly.is_lazy)}, units={anomaly.units}")
    ar.Dataset({"anomaly": anomaly}).to_netcdf(tmp / "anomaly.nc")
    back = ar.open_dataset(tmp / "anomaly.nc")["anomaly"]
    show("anomaly at (lat 0, lon 0), first days", back.isel(lat=10, lon=0, time=slice(0, 4)).values)

    # Reductions of on-disk data stream through a memory budget (here 1 MiB at a time).
    ar.set_stream_bytes(1 << 20)
    show(
        "time-mean air, lat 0..3, lon 0",
        on_disk["air"].mean("time").isel(lon=0, lat=slice(0, 3)).values,
    )
    show(
        "time-std air, lat 0..3, lon 0",
        on_disk["air"].std("time").isel(lon=0, lat=slice(0, 3)).values,
    )
    monthly = on_disk["air"].groupby("time.month").mean()
    show("monthly mean at (lat 10, lon 0)", monthly.isel(lat=10, lon=0).values)

    # Many files as one dataset, combined by their coordinates.
    both = ar.open_mfdataset(str(tmp / "air_*.nc"))
    print("multi-file sizes: " + ", ".join(f"{k}={n}" for k, n in both.sizes.items()))
    show(
        "overall mean (one file, two files)",
        [on_disk["air"].mean().values, both["air"].mean().values],
    )
finally:
    shutil.rmtree(tmp)

//! Disk to disk: compressed netCDF out, lazy reads, streamed computation and reductions.
//!
//! Data opened from files stays on disk: selections, arithmetic, unit conversion and
//! reductions read it chunk by chunk, so it never has to fit in memory. The Python twin is
//! `annoray-python/examples/disk_to_disk.py`; both print the same lines.
//!
//! ```text
//! cargo run --release --example disk_to_disk
//! ```
// annoray's operators return `Result` (never panic), which `arithmetic_side_effects` cannot
// tell; the casts are small loop counters.
#![allow(clippy::cast_precision_loss, clippy::arithmetic_side_effects)]

use std::error::Error;
use std::f64::consts::PI;
use std::path::PathBuf;

use annoray::io::{MfOptions, ToNetcdf, glob, open_dataset, open_mfdataset};
use annoray::ndarray::{Array1, Array3};
use annoray::{DataArray, DataArrayBase, Dataset, Repr, Variable, date_range, isel};

/// Values with three decimals, as the Python twin prints them.
fn show<S: Repr>(label: &str, da: &DataArrayBase<S>) -> Result<(), annoray::CoreError> {
    let v: Vec<String> = da.to_f64()?.iter().map(|v| format!("{v:.3}")).collect();
    println!("{label}: [{}]", v.join(", ").replace("NaN", "nan"));
    Ok(())
}

fn yes(b: bool) -> &'static str {
    if b { "yes" } else { "no" }
}

/// A scratch directory, removed when dropped.
struct Scratch(PathBuf);

impl Drop for Scratch {
    fn drop(&mut self) {
        let _ = std::fs::remove_dir_all(&self.0);
    }
}

/// Two years of daily air temperature on a small grid, in kelvin.
fn dataset() -> Result<Dataset, Box<dyn Error>> {
    // As numpy's linspace(-45, 45, 20): start + i * step, the last value exactly 45.
    let step = 90.0 / 19.0;
    let lat: Vec<f64> = (0..20)
        .map(|i| {
            if i == 19 {
                45.0
            } else {
                f64::from(i) * step - 45.0
            }
        })
        .collect();
    let lon: Vec<f64> = (0..30).map(|i| f64::from(i) * 12.0).collect();
    let air = Array3::from_shape_fn((731, 20, 30), |(t, y, x)| {
        let la = lat.get(y).copied().unwrap_or(f64::NAN);
        let lo = lon.get(x).copied().unwrap_or(f64::NAN);
        288.0 - 30.0 * la.to_radians().sin().abs()
            + 8.0 * (2.0 * PI * t as f64 / 365.25).sin()
            + lo.to_radians().cos()
    });
    let mut ds = Dataset::new();
    let time = date_range("2000-01-01", 731, "D")?;
    ds.add_coord("time", Variable::new(["time"], time.into_dyn())?)?;
    ds.add_coord("lat", Variable::new(["lat"], Array1::from(lat).into_dyn())?)?;
    ds.add_coord("lon", Variable::new(["lon"], Array1::from(lon).into_dyn())?)?;
    let air = Variable::new(["time", "lat", "lon"], air.into_dyn())?.with_unit("K".parse()?);
    ds.add_var("air", air)?;
    // Compressed, chunked netCDF-4.
    let enc = ds.encoding_of_mut("air")?;
    enc.deflate = Some(4);
    enc.chunks = Some(vec![30, 20, 30]);
    Ok(ds)
}

fn main() -> Result<(), Box<dyn Error>> {
    let tmp = Scratch(std::env::temp_dir().join(format!("annoray-example-{}", std::process::id())));
    std::fs::create_dir_all(&tmp.0)?;
    let path = |name: &str| tmp.0.join(name);

    // One file, and one file per year.
    let ds = dataset()?;
    ds.to_netcdf(path("air.nc"))?;
    isel!(ds[time = 0:366])?.to_netcdf(path("air_2000.nc"))?;
    isel!(ds[time = 366:731])?.to_netcdf(path("air_2001.nc"))?;
    println!("wrote air.nc, air_2000.nc, air_2001.nc");

    // Opening reads metadata only; values stay on disk until used.
    let on_disk = open_dataset(path("air.nc"))?;
    let air: DataArray = on_disk.get("air")?.to_owned();
    println!("opened: lazy={}", yes(air.variable().is_lazy()));

    // Arithmetic and unit conversion stay lazy; writing computes chunk by chunk.
    let celsius = air.to_unit(&"degC".parse()?)?;
    let anomaly = (&celsius - &celsius.mean("time")?)?;
    let unit = anomaly.unit().ok_or("no unit")?.to_string();
    println!(
        "anomaly: lazy={}, units={unit}",
        yes(anomaly.variable().is_lazy())
    );
    let mut out = Dataset::new();
    out.add_array("anomaly", anomaly)?;
    out.to_netcdf(path("anomaly.nc"))?;
    let back = open_dataset(path("anomaly.nc"))?;
    let back = back.get("anomaly")?;
    show(
        "anomaly at (lat 0, lon 0), first days",
        &isel!(back[time = 0:4, lat = 10, lon = 0])?,
    )?;

    // Reductions of on-disk data stream through a memory budget (here 1 MiB at a time).
    annoray::set_stream_bytes(1 << 20);
    let mean = air.mean("time")?;
    show(
        "time-mean air, lat 0..3, lon 0",
        &isel!(mean[lat = 0:3, lon = 0])?,
    )?;
    let std = air.std("time")?;
    show(
        "time-std air, lat 0..3, lon 0",
        &isel!(std[lat = 0:3, lon = 0])?,
    )?;
    let monthly = air.groupby("time.month")?.mean()?;
    show(
        "monthly mean at (lat 10, lon 0)",
        &isel!(monthly[lat = 10, lon = 0])?,
    )?;

    // Many files as one dataset, combined by their coordinates.
    let files = glob(path("air_*.nc").to_str().ok_or("path")?)?;
    let both = open_mfdataset(&files, &MfOptions::default())?.dataset;
    let sizes: Vec<String> = both
        .sizes()
        .iter()
        .map(|(k, n)| format!("{k}={n}"))
        .collect();
    println!("multi-file sizes: {}", sizes.join(", "));
    let one = air.mean(..)?.to_f64()?;
    let two = both.get("air")?.mean(..)?.to_f64()?;
    let means: Vec<String> = one.iter().chain(&two).map(|v| format!("{v:.3}")).collect();
    println!("overall mean (one file, two files): [{}]", means.join(", "));
    Ok(())
}

Streaming to a new file#

A file made from a template and filled in hour by hour along an unlimited dimension, read back while it is written, then corrected in place. See also Writing in place.

"""Streaming to a new file: a netCDF-4 file made from a template and filled in as the values
are produced, readable while it is written.

``NetcdfWriter.create`` writes the template's dimensions, variables, attributes, encodings and
coordinates; the data variables read as fill values until written. Each step is appended along
the unlimited ``time`` dimension, a region is rewritten in place, and the file is read back at
any point. The Rust twin is `annoray/examples/stream_to_file.rs`; both print the same lines.

    python examples/stream_to_file.py
"""

import shutil
import tempfile
from pathlib import Path

import numpy as np

import annoray as ar

NLAT, NLON = 4, 6


def show(label, values):
    print(f"{label}: [" + ", ".join(f"{float(v):.2f}" for v in np.ravel(values)) + "]")


def fields(first, hours):
    """`hours` hourly fields of temperature from hour `first`, in kelvin."""
    h = np.arange(first, first + hours)[:, None, None]
    la = np.arange(NLAT)[None, :, None]
    lo = np.arange(NLON)[None, None, :]
    temp = (280.0 + h + 0.5 * la + 0.1 * lo).astype("f4")
    time = np.datetime64("2000-01-01T00", "ns") + np.arange(first, first + hours) * np.timedelta64(
        1, "h"
    )
    return ar.Dataset(
        {"temp": (("time", "lat", "lon"), temp, {"units": "K"})},
        coords={"time": ("time", time)},
    )


tmp = Path(tempfile.mkdtemp(prefix="annoray-stream-"))
try:
    path = tmp / "run.nc"

    # The template: no time steps yet, the grid's coordinates, and how to store the values.
    template = fields(0, 0).assign_coords(
        lat=("lat", -30.0 + 20.0 * np.arange(NLAT)), lon=("lon", 60.0 * np.arange(NLON))
    )
    encoding = {"time": {"units": "hours since 2000-01-01"}, "temp": {"zlib": True, "complevel": 4}}

    with ar.NetcdfWriter.create(path, template, unlimited_dims=["time"], encoding=encoding) as w:
        print(f"created run.nc: time={w.dim_len('time')} (unlimited), lat={NLAT}, lon={NLON}")

        # Each hour's field is appended as soon as it exists.
        for hour in range(6):
            w.append(fields(hour, 1), "time")
            if hour == 2:
                # The file so far, opened lazily while writing goes on.
                now = w.dataset()
                show("after 3 hours, temp at (lat 0, lon 0)", now["temp"].isel(lat=0, lon=0).values)
        print(f"appended 6 hours: time={w.dim_len('time')}")

        # A correction to part of hour 2: the region's own coordinates are left as they are.
        warm = np.full((1, NLAT, 3), 300.0, "f4")
        fix = ar.Dataset({"temp": (("time", "lat", "lon"), warm, {"units": "K"})})
        w.write(fix, region={"time": 2, "lon": slice(3, 6)})
        print("rewrote hour 2 at lon 3..6")

    back = ar.open_dataset(path)
    show("temp at (lat 1, lon 4), hour by hour", back["temp"].isel(lat=1, lon=4).values)
    print(
        f"on disk: time={back.sizes['time']}, units={back['temp'].units}, "
        f"time units={back['time'].encoding['units']}"
    )
finally:
    shutil.rmtree(tmp, ignore_errors=True)

//! Streaming to a new file: a netCDF-4 file made from a template and filled in as the
//! values are produced, readable while it is written.
//!
//! `NetcdfWriter::create` writes the template's dimensions, variables, attributes, encodings
//! and coordinates; the data variables read as fill values until written. Each step is
//! appended along the unlimited `time` dimension, a region is rewritten in place, and the
//! file is read back at any point. The Python twin is
//! `annoray-python/examples/stream_to_file.py`; both print the same lines.
//!
//! ```text
//! cargo run --release --example stream_to_file
//! ```
// annoray's operators return `Result` (never panic), which `arithmetic_side_effects` cannot
// tell; the casts are small loop counters.
#![allow(clippy::cast_precision_loss, clippy::arithmetic_side_effects)]

use std::error::Error;
use std::path::PathBuf;

use annoray::io::{NetcdfWriter, WriteOptions, open_dataset};
use annoray::ndarray::{Array1, Array3};
use annoray::{DataArrayBase, Dataset, Repr, Variable, date_range, isel};

const NLAT: usize = 4;
const NLON: usize = 6;

/// Values with two decimals, as the Python twin prints them.
fn show<S: Repr>(label: &str, da: &DataArrayBase<S>) -> Result<(), annoray::CoreError> {
    let v: Vec<String> = da.to_f64()?.iter().map(|v| format!("{v:.2}")).collect();
    println!("{label}: [{}]", v.join(", "));
    Ok(())
}

/// A scratch directory, removed when dropped.
struct Scratch(PathBuf);

impl Drop for Scratch {
    fn drop(&mut self) {
        let _ = std::fs::remove_dir_all(&self.0);
    }
}

/// `hours` hourly fields of temperature from hour `first`, in kelvin.
fn fields(first: usize, hours: usize) -> Result<Dataset, Box<dyn Error>> {
    let temp = Array3::from_shape_fn((hours, NLAT, NLON), |(h, la, lo)| {
        (280.0 + (first + h) as f64 + 0.5 * la as f64 + 0.1 * lo as f64) as f32
    });
    let start = format!("2000-01-01T{first:02}:00");
    let time = date_range(&start, hours, "h")?;
    let mut ds = Dataset::new();
    ds.add_coord("time", Variable::new(["time"], time.into_dyn())?)?;
    let temp = Variable::new(["time", "lat", "lon"], temp.into_dyn())?.with_unit("K".parse()?);
    ds.add_var("temp", temp)?;
    Ok(ds)
}

fn main() -> Result<(), Box<dyn Error>> {
    let tmp = Scratch(std::env::temp_dir().join(format!("annoray-stream-{}", std::process::id())));
    std::fs::create_dir_all(&tmp.0)?;
    let path = tmp.0.join("run.nc");

    // The template: no time steps yet, the grid's coordinates, and how to store the values.
    let mut template = fields(0, 0)?;
    let lat = Array1::from_iter((0..NLAT).map(|i| -30.0 + 20.0 * i as f64));
    let lon = Array1::from_iter((0..NLON).map(|i| 60.0 * i as f64));
    template.add_coord("lat", Variable::new(["lat"], lat.into_dyn())?)?;
    template.add_coord("lon", Variable::new(["lon"], lon.into_dyn())?)?;
    template.encoding_mut().unlimited_dims = vec!["time".into()];
    template.encoding_of_mut("time")?.time_units = Some("hours since 2000-01-01".into());
    template.encoding_of_mut("temp")?.deflate = Some(4);

    let mut w = NetcdfWriter::create(&path, &template, &WriteOptions::default())?;
    println!(
        "created run.nc: time={} (unlimited), lat={NLAT}, lon={NLON}",
        w.dim_len("time")?
    );

    // Each hour's field is appended as soon as it exists.
    for hour in 0..6 {
        w.append(&fields(hour, 1)?, "time")?;
        if hour == 2 {
            // The file so far, opened lazily while writing goes on.
            let now = w.dataset()?;
            let temp = now.get("temp")?;
            show(
                "after 3 hours, temp at (lat 0, lon 0)",
                &isel!(temp[lat = 0, lon = 0])?,
            )?;
        }
    }
    println!("appended 6 hours: time={}", w.dim_len("time")?);

    // A correction to part of hour 2: the region's own coordinates are left as they are.
    let mut fix = Dataset::new();
    let warm = Array3::<f32>::from_elem((1, NLAT, 3), 300.0);
    fix.add_var(
        "temp",
        Variable::new(["time", "lat", "lon"], warm.into_dyn())?.with_unit("K".parse()?),
    )?;
    w.write(&fix, &[("time".into(), 2), ("lon".into(), 3)])?;
    println!("rewrote hour 2 at lon 3..6");
    w.close()?;

    let back = open_dataset(&path)?;
    let temp = back.get("temp")?;
    show(
        "temp at (lat 1, lon 4), hour by hour",
        &isel!(temp[lat = 1, lon = 4])?,
    )?;
    let time = back.coords().get("time").ok_or("time")?;
    println!(
        "on disk: time={}, units={}, time units={}",
        back.sizes().get("time").copied().unwrap_or(0),
        temp.unit().map_or_else(String::new, ToString::to_string),
        time.encoding().time_units.as_deref().unwrap_or("")
    );
    Ok(())
}
